local player = game.Players.LocalPlayer local mouse = player:GetMouse() local UIS = game:GetService("UserInputService")

-- إنشاء واجهة المستخدم local screenGui = Instance.new("ScreenGui", player:WaitForChild("PlayerGui")) screenGui.Name = "wiliamfly"

local frame = Instance.new("Frame", screenGui) frame.Size = UDim2.new(0, 300, 0, 250) frame.Position = UDim2.new(0.5, -150, 0.5, -125) frame.BackgroundColor3 = Color3.fromRGB(0, 0, 0) frame.Active = true frame.Draggable = true

local darkPurple = Color3.fromRGB(128, 0, 128)

local title = Instance.new("TextLabel", frame) title.Size = UDim2.new(1, 0, 0, 30) title.Text = "FLY GUI V3" title.BackgroundColor3 = Color3.fromRGB(0, 0, 0) title.TextColor3 = darkPurple title.Font = Enum.Font.SourceSansBold title.TextSize = 24

local speedLabel = Instance.new("TextLabel", frame) speedLabel.Position = UDim2.new(0, 10, 0, 40) speedLabel.Size = UDim2.new(0, 280, 0, 20) speedLabel.Text = "Speed: 50" speedLabel.BackgroundTransparency = 1 speedLabel.TextColor3 = darkPurple speedLabel.Font = Enum.Font.SourceSans speedLabel.TextSize = 18

local speed = 50

local function createButton(text, position, callback) local button = Instance.new("TextButton", frame) button.Size = UDim2.new(0, 60, 0, 30) button.Position = position button.Text = text button.BackgroundColor3 = Color3.fromRGB(20, 20, 20) button.TextColor3 = darkPurple button.Font = Enum.Font.SourceSans button.TextSize = 18 button.MouseButton1Click:Connect(callback) return button end

local up = createButton("Up", UDim2.new(0, 10, 0, 70), function() if root then root.Velocity = Vector3.new(0, speed, 0) end end)

local down = createButton("Down", UDim2.new(0, 80, 0, 70), function() if root then root.Velocity = Vector3.new(0, -speed, 0) end end)

local plus = createButton("+", UDim2.new(0, 150, 0, 70), function() speed = speed + 10 speedLabel.Text = "Speed: " .. speed end)

local minus = createButton("-", UDim2.new(0, 220, 0, 70), function() speed = math.max(10, speed - 10) speedLabel.Text = "Speed: " .. speed end)

local flying = false local root = nil

local flyButton = createButton("Fly [F]", UDim2.new(0.5, -30, 0, 110), function() flying = not flying end)

-- زر إغلاق الواجهة local X = createButton("X", UDim2.new(1, -30, 0, 0), function() screenGui:Destroy() end) X.Size = UDim2.new(0, 30, 0, 30) X.Position = UDim2.new(1, -30, 0, 0)

-- التحكم في الطيران UIS.InputBegan:Connect(function(input) if input.KeyCode == Enum.KeyCode.F then flying = not flying end end)

game:GetService("RunService").RenderStepped:Connect(function() if flying and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then root = player.Character.HumanoidRootPart local camCF = workspace.CurrentCamera.CFrame local moveVector = Vector3.new()

if UIS:IsKeyDown(Enum.KeyCode.W) then
        moveVector = moveVector + camCF.LookVector
    end
    if UIS:IsKeyDown(Enum.KeyCode.S) then
        moveVector = moveVector - camCF.LookVector
    end
    if UIS:IsKeyDown(Enum.KeyCode.A) then
        moveVector = moveVector - camCF.RightVector
    end
    if UIS:IsKeyDown(Enum.KeyCode.D) then
        moveVector = moveVector + camCF.RightVector
    end

    root.Velocity = moveVector.Unit * speed
end

end)
